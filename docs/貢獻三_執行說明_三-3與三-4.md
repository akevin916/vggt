### 3.3 執行與匯出

```bash
torchrun --nproc_per_node=1 pipeline/training/launch.py --config c3_reltile_b25_A
python export_relmask.py logs/c3_reltile_b25_A/ckpts/last.pt outputs/colmask 0.25
```

```python
# export_relmask.py <ckpt> <out_dir> <rho>（一次性，不進 repo）
# 輸出 relmask_b25.npy（學到的）、relrand_b25_s{0,1,2}.npy（每桶保留數相同、位置隨機）、reluniform_b25.npy（每桶都留 25%）
import sys, numpy as np, torch
th = torch.load(sys.argv[1], map_location="cpu")["model"]["aggregator.colgate.theta"].numpy()   # [6,8,8]
out, rho = sys.argv[2], float(sys.argv[3]); tag = f"b{round(rho*100)}"
ub = [1, 2, 4, 8, 16]; S = 64
def bucket(d): return sum(d > u for u in ub)
w = np.zeros(6)                                   # 每桶在 c64 的 (query 幀, key 幀≥1) 配對數
for s in range(S):
    for t in range(1, S):
        if s != t: w[bucket(abs(s - t))] += 1
# 每格的權重 = 該桶配對數 × 這格在 30×37 上出現的次數
yy, xx = np.arange(30) % 8, np.arange(37) % 8
cnt = np.zeros((8, 8)); np.add.at(cnt, (yy[:, None].repeat(37, 1), xx[None, :].repeat(30, 0)), 1)
cell_w = w[:, None, None] * cnt[None]             # [6,8,8]
order = np.argsort(-th.ravel()); acc = np.cumsum(cell_w.ravel()[order]) / cell_w.sum()
keep = np.zeros(th.size, bool); keep[order[:np.searchsorted(acc, rho) + 1]] = True
keep = keep.reshape(6, 8, 8); np.save(f"{out}/relmask_{tag}.npy", keep)
print("每桶保留比例", keep.reshape(6, -1).mean(1).round(3), "加權總比例", (cell_w * keep).sum() / cell_w.sum())
for s in range(3):
    rng = np.random.default_rng(s); r = np.zeros_like(keep)
    for b in range(6): r[b].flat[rng.choice(64, keep[b].sum(), replace=False)] = True
    np.save(f"{out}/relrand_{tag}_s{s}.npy", r)
u = np.zeros_like(keep)
for b in range(6): u[b].flat[np.argsort(-th[b].ravel())[:round(64 * rho)]] = True
np.save(f"{out}/reluniform_{tag}.npy", u)
```
